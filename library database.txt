SELECT b.title,b.publisher_name,ba.author_name,bc.noofcopies
FROM book b
JOIN book_author ba ON b.book_id = ba.book_id
JOIN book_copies bc ON b.book_id = bc.book_id;

#2
SELECT card_no,COUNT(book_id) AS books_borrowed
FROM book_lending
WHERE date_out BETWEEN '2025-03-05' AND '2025-3-10'
GROUP BY card_no   

+
HAVING COUNT(*) >1;
